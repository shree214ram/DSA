/**
 * @param {number} numCourses
 * @param {number[][]} prerequisites
 * @return {boolean}
 */
var canFinish = function (numCourses, prerequisites) {
    // Prepare adjacency list (Your exact logic)
    const adj = new Map();
    for (let i = 0; i < prerequisites.length; i++) {
        const course = prerequisites[i][0];
        const prereq = prerequisites[i][1];
        adj.set(course, adj.get(course) ? [...adj.get(course), prereq] : [prereq]);
    }

    // Track fully processed nodes globally
    const visited = Array(numCourses).fill(false);
    // NEW: Track nodes in the CURRENT recursion path
    const inStack = Array(numCourses).fill(false); 

    const dfs = (course) => {
        // 1. If it's already in the current path, we found a loop/circle!
        if (inStack[course]) return true;
        
        // 2. If it's already fully processed and safe, no need to check again
        if (visited[course]) return false;

        // Add to current path and mark as visited
        inStack[course] = true;
        visited[course] = true;

        // Explore neighbors
        const pre = adj.get(course);
        if (pre) {
            for (let i = 0; i < pre.length; i++) {
                // If any neighbor path finds a circle, return true immediately
                if (dfs(pre[i])) return true; 
            }
        }

        // BACKTRACK: Remove from current path before returning
        inStack[course] = false;
        return false; // No circle found along this path
    };

    // Call DFS for each course (Your exact logic)
    for (let i = 0; i < numCourses; i++) {
        if (dfs(i)) {
            return false; // If a cycle is found, we cannot finish all courses
        }
    }

    return true; // No cycles found anywhere, we can finish!
};
