/**
 * @param {number} numCourses
 * @param {number[][]} prerequisites
 * @return {boolean}
 */
var canFinish = function(numCourses, prerequisites) {
    if(prerequisites.length==0){
        return true
    }
    let num=0
    const hp={}
    for(let i=0;i<prerequisites.length;i++){
        const start=prerequisites[i]
        const starting=start[0];
        const ending=start[1];
        const completeKey=starting+"-"+ending;
        const completeRevKey=ending+"-"+starting;
        hp[completeKey]=false;
        hp[completeRevKey]=false;
    }
    const DfsUtil=(index)=>{
        if(index>=prerequisites.length){
            return 
        }
        const start =prerequisites[index];
        const starting=start[0];
        const ending=start[1];
        const completeKey=starting+"-"+ending;
        const completeRevKey=ending+"-"+starting;
        if((hp[completeKey]==false && hp[completeRevKey] ==false)){
             hp[completeKey]=true;
            hp[completeRevKey]=true;
            num++
             DfsUtil(index+1)
        }else {
            return false 
        }
        return num==numCourses
    }
    return DfsUtil(0)
};
