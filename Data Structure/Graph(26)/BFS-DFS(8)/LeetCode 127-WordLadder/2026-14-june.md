            const top=q[0]
            q.shift();

            for(let j=0;j<oldWord.length;j++){
                for(let k=0;k<char.length;k++){
                    const orignal=oldWord[j]

                    if(newWord == endWord) return lev+1
                    if(hp.has(newWord)){

                        q.push(newWord)

                }
            }
        }
        lev++
    }
    return lev
