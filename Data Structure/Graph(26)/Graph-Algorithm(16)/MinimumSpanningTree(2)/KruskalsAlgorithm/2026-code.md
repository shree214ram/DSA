<script>
class DisjointSet {
  constructor(size) {
    this.parent = new Int32Array(size);
    this.rank = new Int32Array(size);
    for (let i = 0; i < size; i++) {
      this.parent[i] = i;
    }
  }

  // Find with Path Compression: O(α(N))
  find(i) {
    let root = i;
    while (root !== this.parent[root]) {
      root = this.parent[root];
    }
    // Compress path
    let curr = i;
    while (curr !== root) {
      let nxt = this.parent[curr];
      this.parent[curr] = root;
      curr = nxt;
    }
    return root;
  }

  // Union by Rank: O(α(N))
  union(i, j) {
    let rootI = this.find(i);
    let rootJ = this.find(j);

    if (rootI !== rootJ) {
      if (this.rank[rootI] < this.rank[rootJ]) {
        this.parent[rootI] = rootJ;
      } else if (this.rank[rootI] > this.rank[rootJ]) {
        this.parent[rootJ] = rootI;
      } else {
        this.parent[rootJ] = rootI;
        this.rank[rootI]++;
      }
      return true; // Successfully merged
    }
    return false; // Already connected (creates a cycle)
  }
}

/**
 * Kruskal's Algorithm to find Minimum Spanning Tree (MST)
 * @param {number} numVertices - Total number of vertices (0 to numVertices - 1)
 * @param {Array<[number, number, number]>} edges - Array of [u, v, weight]
 * @returns {{ mst: Array<[number, number, number]>, minCost: number }}
 */
function kruskalMST(numVertices, edges) {
  // 1. Sort all edges by weight ascending: O(E log E)
  edges.sort((a, b) => a[2] - b[2]);

  const dsu = new DisjointSet(numVertices);
  const mst = [];
  let minCost = 0;

  // 2. Iterate through sorted edges and pick the lightest non-cyclic ones
  for (let i = 0; i < edges.length; i++) {
    const [u, v, weight] = edges[i];

    // If adding this edge doesn't create a cycle
    if (dsu.union(u, v)) {
      mst.push([u, v, weight]);
      minCost += weight;

      // Early exit optimization: An MST with V vertices always has V - 1 edges
      if (mst.length === numVertices - 1) break;
    }
  }

  return { mst, minCost };
}

// ==================== Example Usage ====================
const numVertices = 4;
// [u, v, weight]
const edges = [
  [0, 1, 10],
  [0, 2, 6],
  [0, 3, 5],
  [1, 3, 15],
  [2, 3, 4]
];

const result = kruskalMST(numVertices, edges);
console.log("Total MST Cost:", result.minCost); // Output: 19
console.log("MST Edges:", result.mst);         // Output: [[2, 3, 4], [0, 3, 5], [0, 1, 10]]
