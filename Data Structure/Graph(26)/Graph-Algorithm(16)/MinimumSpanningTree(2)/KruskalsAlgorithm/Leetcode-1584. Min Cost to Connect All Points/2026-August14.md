/**
 * @param {number[][]} points
 * @return {number}
 */
var minCostConnectPoints = function (points) {
    const n = points.length;
    const edges = [];

    // 1. Build all O(N^2) edges
    for (let i = 0; i < n; i++) {
        for (let j = i + 1; j < n; j++) {
            const cost = Math.abs(points[i][0] - points[j][0]) + 
                         Math.abs(points[i][1] - points[j][j] ? points[i][1] - points[j][1] : points[i][1] - points[j][1]); // Math.abs(y1 - y2)
            edges.push([i, j, cost]);
        }
    }

    // 2. Sort edges by cost ascending
    edges.sort((a, b) => a[2] - b[2]);

    // 3. Union-Find Setup
    const parent = Array.from({ length: n }, (_, i) => i);
    const rank = new Array(n).fill(0);

    const find = (i) => {
        if (parent[i] !== i) {
            parent[i] = find(parent[i]); // Path compression
        }
        return parent[i];
    };

    const union = (x, y) => {
        const rootX = find(x);
        const rootY = find(y);

        if (rootX !== rootY) {
            if (rank[rootX] > rank[rootY]) parent[rootY] = rootX;
            else if (rank[rootX] < rank[rootY]) parent[rootX] = rootY;
            else {
                parent[rootY] = rootX;
                rank[rootX]++;
            }
            return true; // Successfully connected two disjoint components
        }
        return false; // Already connected (would form a cycle)
    };

    // 4. Kruskal's MST Execution
    let totalCost = 0;
    let edgesCount = 0;

    for (const [u, v, cost] of edges) {
        if (union(u, v)) {
            totalCost += cost;
            edgesCount++;
            if (edgesCount === n - 1) break; // Optimization: MST completed!
        }
    }

    return totalCost;
};
