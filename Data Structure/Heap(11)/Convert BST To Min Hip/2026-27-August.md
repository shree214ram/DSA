```javascript
 * Definition for a binary tree node.
 * function TreeNode(val, left, right) {

 *     this.right = (right===undefined ? null : right)

/**
 * @param {TreeNode} root
 * @return {TreeNode}
 */
var convertBSTToMinHeap = function(root) {
    const sortedValues = [];
    
    // Step 1: Inorder Traversal to extract BST values in sorted order
    function inorder(node) {
        if (!node) return;
        inorder(node.left);
        sortedValues.push(node.val);
        inorder(node.right);
    }
    
    // Step 2: Preorder Traversal to overwrite tree nodes in Min-Heap order (parent < left < right)
    let index = 0;
    function preorder(node) {
        if (!node) return;
        
        node.val = sortedValues[index++];
        preorder(node.left);
        preorder(node.right);
    }
    
    inorder(root);
    preorder(root);
    
    return root;
};

```

---

### Complexity Analysis

* **Time Complexity:** $\mathcal{O}(N)$ — One Inorder pass to collect sorted values plus one Preorder pass to overwrite values.
* **Space Complexity:** $\mathcal{O}(N)$ — Extra space for the `sortedValues` array to store $N$ node values, plus $\mathcal{O}(H)$ for the recursion call stack (where $H$ is the tree height).
