var minWindow = function (s, t) {
    if (s.length < t.length) return "";
    
    const tM = new Map();
    const sM = new Map();

    // 1. Map out target requirements
    for (let i = 0; i < t.length; i++) {
        const current = t[i].charCodeAt(0);
        tM.set(current, (tM.get(current) || 0) + 1);
    }

    let count = 0;
    let minL = Infinity;
    let start = 0;
    let minI = 0;

    // 2. Scan string s in a SINGLE continuous loop
    for (let i = 0; i < s.length; i++) {
        const current = s[i].charCodeAt(0);
        sM.set(current, (sM.get(current) || 0) + 1);
        
        // Only increment count if we actually need more of this character
        if (tM.has(current) && sM.get(current) <= tM.get(current)) {
            count++;
        }

        // 3. Shrink window when valid
        if (count === t.length) {
            let startChar = s[start].charCodeAt(0);
            
            while (!tM.has(startChar) || sM.get(startChar) > tM.get(startChar)) {
                sM.set(startChar, sM.get(startChar) - 1);
                start++; // Increment after adjusting map
                startChar = s[start].charCodeAt(0);
            }

            // Update minimum snapshot
            const currentL = i - start + 1;
            if (currentL < minL) {
                minL = currentL;
                minI = start;
            }
        }
    }
    
    return minL === Infinity ? "" : s.substring(minI, minI + minL);
};
