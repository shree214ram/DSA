/**
 * @param {string} text1
 * @param {string} text2
 * @return {number}
 */
var longestCommonSubsequence = function (text1, text2) {
    const s1 = text1;
    const s2 = text2;
    if(s1==s2) return s1.length
    if (s1.length == 1 && s2.length == 1) return s1[0] == s2[0] ? 1 : 0
    const DP =  Array.from({ length: s1.length + 1 }, () => Array(s2.length + 1).fill(0))
    for (let i = 1; i < DP.length; i++) {
        for (let j = 1; j < DP[0].length; j++) {
            // console.log(s1[i-1],"first")
            // console.log(s2[j-1],"second")
            if(s1[i-1]===s2[j-1]){
                // console.log("hay")
                DP[i][j]=1+DP[i-1][j-1]
            }else{
                DP[i][j]=Math.max(DP[i-1][j],DP[i][j-1])
            }
        }
    }
    // console.log(DP,"DP")
    return DP[DP.length-1][DP[0].length-1]
};
